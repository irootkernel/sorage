_sorage() {
  local cur commands option_words
  cur="${COMP_WORDS[COMP_CWORD]}"
  commands="accept archive backup channel comment completion config daemon decline delete doctor events fetch get inbox init mcp memo outbox pin preflight processing project review revise send token unarchive uninstall unpin vault version web withdraw"
  option_words="--as --as-user --confirm --cursor --expected-row-version --json --limit"
  case "${COMP_WORDS[1]}" in
    send) option_words="$option_words --to --title --file --body --supersedes --allow-external-source --allow-unregistered --idempotency-key --replay-only" ;;
  esac
  if [[ "${COMP_CWORD}" -eq 1 ]]; then
    COMPREPLY=( $(compgen -W "${commands}" -- "${cur}") )
  else
    COMPREPLY=( $(compgen -W "${option_words}" -- "${cur}") )
  fi
  return 0
}

complete -F _sorage sorage
