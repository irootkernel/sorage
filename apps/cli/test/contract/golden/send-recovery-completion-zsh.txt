#compdef sorage

_sorage() {
  local -a commands
  commands=(
    'version:print the product version'
    'preflight:inspect the selected installation without initializing or opening it'
    'init:create the Sorage installation: home tree, configuration, database, and Vault'
    'uninstall:remove the Sorage installation while keeping the Vault'
    'config:inspect and change the installation configuration'
    'config show:print the effective configuration exactly as declared in config.yaml'
    'config validate:validate config.yaml against the schema and report every issue'
    'config set:set one configuration leaf through the atomic comment-preserving store'
    'config edit:open config.yaml in $EDITOR and adopt the result through the atomic store'
    'doctor:run the milestone-scoped installation checks and report the catalog'
    'memo:record and manage Project reminders separately from Handoffs'
    'memo add:record an open Memo'
    'memo list:list Project Memos, open by default'
    'memo show:show one complete Memo'
    'memo update:edit an open Memo at its observed Row Version'
    'memo done:mark a Memo handled'
    'memo dismiss:dismiss a Memo without claiming work was done'
    'memo reopen:reopen a closed Memo without executing its content'
    'channel:manage attributed agent communication Channels for one orchestration invocation'
    'channel create:create an open Channel with its single main participant'
    'channel add:register a worker or judge participant on an open Channel'
    'channel remove:remove a participant idempotently; reversible by re-adding the handle'
    'channel post:post one attributed message upstream through the shared content interface'
    'channel messages:list messages with full inline bodies; unhandled oldest first by default'
    'channel ack:record the User handling decision for the named messages'
    'channel close:terminal idempotent close; freezes posting, keeps reads and acks'
    'channel list:list Channels with both identifiers'
    'channel show:show one Channel with its complete participant roster and counts'
    'project:register and inspect Projects and their bindings'
    'project add:register a Project and its first directory binding'
    'project list:list every Project with its binding count, flagging unbound Projects'
    'project show:show one Project with every binding, its counts, and its lifecycle state'
    'project rename:change the display name; the slug is identity and is never renamed'
    'project bind:add another directory binding to a registered Project'
    'project unbind:remove a directory binding; --confirm is required when open Handoffs would lose their binding'
    'project rebind:replace one recorded Project binding with an existing directory'
    'project archive:archive a Project: existing Handoffs remain available, new sends and receipts are refused'
    'project unarchive:return an archived Project to active'
    'project resolve:print what the actor resolver would decide for a path, or for --as <project-slug>'
    'send:hand one current document to one or more recipient Projects, one Handoff each'
    'inbox:list the Handoffs sent to the resolved recipient Project'
    'outbox:list the Handoffs the resolved sender sent'
    'get:read one Handoff'\''s metadata; records nothing'
    'fetch:read the current Artifact'\''s metadata and local path; the recipient'\''s first fetch is recorded'
    'events:read the bounded recent metadata timeline; records nothing'
    'review:read, create, withdraw, and remove Review Notes'
    'review show:read the current Review Note; records nothing'
    'review set:create or update this Handoff'\''s Review Note as the recipient or the User'
    'review withdraw:withdraw this Handoff'\''s current Review Note as the recipient, whichever actor authored it'
    'review remove:remove this Handoff'\''s Review Note administratively; a User-admin operation'
    'revise:replace the current Artifact or resolve the Review Note without changing content'
    'accept:accept the exact Revision as the recipient, closing the exchange'
    'decline:decline the Handoff as the recipient with a recorded reason'
    'withdraw:withdraw the Handoff as the sender before the recipient has engaged'
    'pin:pin the Handoff against deletion; a User-admin operation'
    'unpin:release a pinned Handoff; a User-admin operation'
    'archive:archive a terminal Handoff out of the default listings; a User-admin operation'
    'unarchive:return an archived Handoff to the default listings; a User-admin operation'
    'delete:request and decide Handoff deletion'
    'delete request:ask the User to delete this Handoff in any non-deleted state'
    'delete approve:approve a pending deletion as the User; the tombstone keeps no content'
    'delete reject:reject a pending deletion as the User; the Handoff is retained'
    'vault:inspect, verify, and relocate the Artifact Vault'
    'vault status:report the Vault path, marker, counts, and sizes'
    'vault verify:run the Git-independent integrity sweep and name every finding; never repairs'
    'vault move:relocate the Vault under vault-move.lock; every other mutation pauses while it runs'
    'backup:verify, run, and restore Vault backups'
    'backup run:run one backup now: export the snapshot, commit managed changes, and report each outcome'
    'backup status:report the backup history from backup_runs: last attempt, success, commit, push, and failure'
    'backup verify:run every vault verify check plus the Git configuration and snapshot consistency checks; never repairs'
    'backup enable:turn the daily backup schedule on with its local time and zone'
    'backup disable:turn the daily backup schedule off'
    'backup enable-push:turn remote push on for one configured remote and branch; the push is a fast-forward only'
    'backup disable-push:turn remote push off; local commits continue'
    'backup restore:rebuild an empty installation from a Vault backup copy, adopting its installationId; the daemon must be stopped'
    'completion:print the shell completion script for zsh or bash; source it from the shell'\''s profile'
    'daemon:run and control the local daemon'
    'daemon start:start the daemon and confirm it through health'
    'daemon stop:stop the daemon after a graceful drain of in-flight requests'
    'daemon restart:stop and start again; required after a server.host or server.port change'
    'daemon status:read run/daemon.json and verify installationId through health'
    'daemon serve:internal entry point that runs the daemon in the foreground until stopped'
    'web:start the daemon when needed and open the Web control plane with a one-time session secret'
    'mcp:serve the STDIO MCP adapter whose tools mirror the messaging CLI (interface section 39)'
    'token:manage the Installation API token'
    'token rotate:replace the Installation API token, invalidating every live browser session'
  )
  _describe -t commands 'command' commands
  case "$words[2]" in
    send) _arguments '--as' '--as-user' '--confirm' '--expected-row-version' '--limit' '--cursor' '--json' '--to' '--title' '--file' '--body' '--supersedes' '--allow-external-source' '--allow-unregistered' '--idempotency-key' '--replay-only'; return ;;
  esac
  _arguments \
      '--as'
      '--as-user'
      '--confirm'
      '--expected-row-version'
      '--limit'
      '--cursor'
      '--json'
}

if (( $+functions[compdef] )); then
  compdef _sorage sorage
fi
